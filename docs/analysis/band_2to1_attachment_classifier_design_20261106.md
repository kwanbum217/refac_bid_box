### 3.4 미확인을 줄이는 방법

1. **형식 판별을 매직 단독에서 매직+확장자+OLE 스트림으로 넓힙니다.** `Workbook` 스트림이면 XLS로 보내고, `Contents/content.hpf` 가 있으면 HWPX로 보냅니다. HWP 1건이 바로 해결됩니다.
2. **PDF 손상은 재다운로드·복구 순서로 처리합니다.** `pypdfium2` 로 열리는지 먼저 보고, 실패하면 `%%EOF` 이후를 잘라 재시도합니다. 두 건은 같은 공고의 HWP 에서 이미 문구를 확인했으므로 판별에는 영향이 없습니다.
3. **XLSX·ZIP 은 판별 대상에서 명시적으로 제외하고 사유를 남깁니다.** 판별 입력은 공고문·과업지시서·표준공고문으로 한정하고, 내역서는 값 추출 단계로 미룹니다.
4. **문구 사전을 넓힙니다.** 이번에 놓친 표기는 `학술용역 적격심사`, `제2장 2`, `제2장의 2`, `별표2`, `낙찰자선정 세부기준`, `[별표 1] 추정가격 2억원 미만 1억원 이상` 입니다. R1·R3에 별칭을 추가하면 미검출 15·8건이 줄어듭니다.
5. **HTTP `000` 은 1회 재시도합니다.** 1초 예절 대기를 지키면 2건 모두 1회 실패했습니다. 같은 공고의 다른 첨부가 있어 판별에는 영향이 없었지만, 단일 첨부 공고에서는 치명적입니다.

def download(best: dict[str, dict]) -> None:
    for no, r in sorted(best.items()):
        seen = set()
        for slot, key in (("std", "std_url"), ("f1", "u1"), ("f2", "u2"), ("f3", "u3")):
            url = r.get(key)
            if not url or url in seen:
                continue
            seen.add(url)
            out = RAW / no / f"{slot}.bin"
            out.parent.mkdir(parents=True, exist_ok=True)
            if out.exists() and out.stat().st_size > 0:
                continue
            subprocess.run(
                ["curl", "-sL", "--max-time", "30", "-o", str(out),
                 "-w", "%{http_code} %{size_download} %{time_total}", url],
                capture_output=True, text=True, check=False,
            )
            time.sleep(1.05)

            return "hwpx"
        if any(n.startswith("xl/") for n in names):
            return "xlsx"
        return "zip"
    if data.startswith(b"<?xml"):
        return "hwpml"
    if data.startswith(OLE):
        return "hwp"
    if b"<html" in data[:4000].lower():
        return "html"
    return "unknown"


def extract_all() -> list[dict]:
    TEXT.mkdir(parents=True, exist_ok=True)
    records = []
    for d in sorted(p for p in RAW.iterdir() if p.is_dir()):
        for f in sorted(d.glob("*.bin")):
            data = f.read_bytes()
            kind = sniff(data)
            t0 = time.perf_counter()
            text, reason = "", None
            if kind in ("hwp", "hwpx", "hwpml"):
                doc = extract_document_text(data)
                text = doc.flat if doc.ok else ""
                reason = None if doc.ok else doc.reason
            elif kind == "pdf":
                try:
                    with pdfplumber.open(io.BytesIO(data)) as pdf:
                        if len(pdf.pages) > MAX_PDF_PAGES:
                            raise ValueError(f"PDF 쪽수 초과({len(pdf.pages)})")
                        text = "\n".join((p.extract_text() or "") for p in pdf.pages)
                    if not text.strip():
                        reason = "PDF 텍스트 레이어 없음"
                except Exception as exc:
                    reason = f"PDF 추출 실패: {type(exc).__name__}"
            else:
                reason = f"{kind} 미지원(조사 범위 제외)"
            ms = (time.perf_counter() - t0) * 1000
            records.append({
                "bid_no": d.name, "slot": f.stem, "bytes": len(data), "kind": kind,
                "ok": bool(text.strip()), "chars": len(text), "reason": reason,
                "extract_ms": round(ms, 2),
            })
            if text.strip():
                (TEXT / f"{d.name}__{f.stem}.txt").write_text(text, encoding="utf-8")
    json.dump(records, open(CACHE / "extraction.json", "w"), ensure_ascii=False, indent=1)
    return records


def main() -> None:
    ap = argparse.ArgumentParser()
    ap.add_argument("--download", action="store_true")
    args = ap.parse_args()

    best = population()
    if args.download:
        download(best)
    records = extract_all()
    by_bid = defaultdict(list)
    for rec in records:
        by_bid[rec["bid_no"]].append(rec)

    truth = {b: "MOIS_ACADEMIC" for b in best}
    for label, bids in LABELS.items():
        for b in bids:
            truth[b] = label

    texts = {}
    for b in best:
        chunks = []
        for rec in by_bid.get(b, []):
            p = TEXT / f"{b}__{rec['slot']}.txt"
            if p.exists():
                chunks.append(norm(p.read_text(errors="ignore")))
        texts[b] = norm(" ".join(chunks))

    print("== 모집단 ==")
    print("rows", len(json.load(open(CACHE / "population.json"))), "unique", len(best))
    print("files", len(records), "ok", sum(r["ok"] for r in records),
          "kinds", dict(Counter(r["kind"] for r in records)))
    print("failed reasons", dict(Counter(r["reason"] for r in records if not r["ok"])))
    ms = sorted(r["extract_ms"] for r in records)
    print("extract_ms p50", ms[len(ms) // 2], "p95", ms[int(len(ms) * .95)], "max", ms[-1])

    print("\n== 패턴별 적중·오탐 ==")
    print("id\tbranch\tmatched\thit\tfp\tmiss")
    for pid, branch, pat in PATTERNS:
        rx = re.compile(pat)
        matched = {b for b, t in texts.items() if t and rx.search(t)}
        hit = {b for b in matched if truth[b] in TARGET[branch]}
        fp = matched - hit
        miss = {b for b in truth if truth[b] in TARGET[branch] and b not in matched}
        print(pid, branch, len(matched), len(hit), len(fp), len(miss), sep="\t")

    def classify(b: str) -> str:
        if not by_bid.get(b):
            return "NO_ATTACH"
        if not texts[b]:
            return "EXTRACT_FAIL"
        text = texts[b]
        weight = {
            "MOIS_ACADEMIC": [("R1", 2.0), ("R2", 2.0), ("R3", 2.0), ("R4", 2.0), ("R5", 0.2)],
            "PPS": [("R6", 2.0), ("R7", 2.0)],
            "MOIS4": [("R8", 2.0), ("R9", 1.5)],
            "LOCAL": [("R10", 2.0), ("R11", 1.5)],
            "MOIS_OTHER": [("R12", 2.0)],
        }
        pmap = {pid: re.compile(p) for pid, _, p in PATTERNS}
        scores = defaultdict(float)
        for branch, items in weight.items():
            for pid, w in items:
                for m in pmap[pid].finditer(text):
                    win = text[max(0, m.start() - 160): m.end() + 160]
                    if not VERB.search(win):
                        continue
                    mult = 0.25 if NOISE.search(win) else 1.0
                    if DECISIVE.search(win):
                        mult *= 1.5
                    scores[branch] += w * mult
        order = ["MOIS4", "PPS", "LOCAL", "MOIS_OTHER", "MOIS_ACADEMIC"]
        if not scores or max(scores.values()) <= 0:
            return "UNSPECIFIED"
        branch = max(order, key=lambda k: (scores[k], -order.index(k)))
        if branch == "MOIS_ACADEMIC" and re.search(r"87\.745", text):
            branch = "MOIS_OTHER"
        return branch

    pred = {b: classify(b) for b in best}
    print("\n== 최종 판정 ==")
    print("truth", dict(Counter(truth.values())))
    print("pred ", dict(Counter(pred.values())))
    print("correct", sum(truth[b] == pred[b] for b in best), "/", len(best))
    print("errors", [(b, truth[b], pred[b]) for b in best if truth[b] != pred[b]])
    print("자동 판별(판정기)", sum(1 for b in best if pred[b] not in ("UNSPECIFIED", "NO_ATTACH")))
    print("자동 판별(정답 기준)", sum(1 for b in best if truth[b] not in ("UNSPECIFIED", "NO_ATTACH")))


if __name__ == "__main__":
    main()
